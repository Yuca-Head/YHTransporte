    public async Task<IEnumerable<(int ThirdPartyId, IEnumerable<Address> Addresses)>>
    GetAddressesFromThirdParties(
        IEnumerable<int> thirdPartyIds,
        CancellationToken cancellationToken = default)
    {
        using var connection = _factory.Create();

        var command = new CommandDefinition(
            """
            SELECT
                tpa.IdThirdParty,
                a.Id,
                a.Details,
                a.IdMunicipality,
                a.MunicipalityName,
                a.DepartmentId,
                a.DepartmentName
            FROM ThirdPartiesAddresses tpa
            INNER JOIN vw_ThirdPartyAddress a   
                ON a.Id = tpa.IdThirdParty
            WHERE tpa.IdThirdParty IN @ThirdPartyIds;
            """,
            new { ThirdPartyIds = thirdPartyIds },
            cancellationToken: cancellationToken);

        var rows = await connection.QueryAsync<ThirdPartyAddressSqlDto>(command);

        return rows
            .GroupBy(x => x.IdThirdParty)
            .Select(group => (
                ThirdPartyId: group.Key,
                Addresses: group.Select( x => new Address(x.Details, 
                new Municipality(x.MunicipalityName, new Department(x.DepartmentName)
                {Key = x.DepartmentId}){Key = x.IdMunicipality}){Key = x.Id})
            ));
    }
